Step 1 — Prepare Local Server

sudo apt update

sudo mkdir -p /var/www/meat-lovers-cims

cd /var/www/meat-lovers-cims

Step 2 — Configure Environment
cp .env.example .env
Set:

DB_DATABASE=meat_lovers_cims
DB_USERNAME=meat_lovers_user
DB_PASSWORD=StrongLocalPassword
JWT_SECRET=CHANGE_THIS_LOCAL_SECRET
APP_URL=http://SERVER_LAN_IP
Each frontend app must have:
VITE_API_BASE_URL=http://SERVER_LAN_IP/api

apps/website/.env

3. Database Setup
sudo mysql
CREATE DATABASE meat_lovers_cims;
CREATE USER 'meat_lovers_user'@'localhost' IDENTIFIED BY 'StrongLocalPassword';
GRANT ALL PRIVILEGES ON meat_lovers_cims.* TO 'meat_lovers_user'@'localhost';
FLUSH PRIVILEGES;
EXIT;
Run all migrations in strict order:
for file in database/migrations/*.sql; do
mysql -u meat_lovers_user -p meat_lovers_cims < "$file"
done
Run all seeds in strict order:
for file in database/seeds/*.sql; do
mysql -u meat_lovers_user -p meat_lovers_cims < "$file"
done
No migration may be skipped.
