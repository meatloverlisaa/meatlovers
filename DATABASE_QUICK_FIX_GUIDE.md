## 🔍 Quick Database Health Check

```bash
cd /home/the-macharias/MeatLovers/meetlovers/api
node -e "
const { PrismaClient } = require('@prisma/client');
const prisma = new PrismaClient();

async function quickCheck() {
  // Check missing profiles
  const noProfile = await prisma.user.count({
    where: { role: { notIn: ['SUPER_ADMIN', 'ADMIN'] }, employee_profile: null }
  });
  
  // Check expired tokens
  const expiredTokens = await prisma.refreshToken.count({
    where: { expires_at: { lt: new Date() }, is_revoked: false }
  });
  
  console.log('Users without profiles:', noProfile);
  console.log('Expired tokens not revoked:', expiredTokens);
  console.log(noProfile === 0 && expiredTokens === 0 ? '✅ ALL GOOD' : '❌ NEEDS FIXING');
  
  await prisma.\$disconnect();
}

quickCheck();
"
```
